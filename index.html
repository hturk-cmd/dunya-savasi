async function loadMap() { $("mapStatus").textContent = "regions.geojson okunuyor..."; const response = await fetch("./regions.geojson"); if (!response.ok) { throw new Error( "regions.geojson yüklenemedi. Dosyanın index.html ile aynı klasörde olduğuna emin ol." ); } const data = await response.json(); if ( data.type !== "FeatureCollection" || !Array.isArray(data.features) ) { throw new Error("GeoJSON dosyasının biçimi geçersiz."); } features = data.features.filter(f => f && f.type === "Feature" && f.geometry && ["Polygon", "MultiPolygon"].includes(f.geometry.type) ); if (!features.length) { throw new Error("Çizilebilir bölge bulunamadı."); } // GeoJSON poligonlarının yönünü kontrol et. // Ters yöndeki poligonlar dünya küresi gibi görünebilir. for (const feature of features) { const geometry = feature.geometry; const polygons = geometry.type === "Polygon" ? [geometry.coordinates] : geometry.coordinates; for (const polygon of polygons) { const polygonFeature = { type: "Feature", properties: {}, geometry: { type: "Polygon", coordinates: polygon } }; if (d3.geoArea(polygonFeature) > 2 * Math.PI) { for (const ring of polygon) { ring.reverse(); } } } } $("regionCount").textContent = features.length.toLocaleString("tr-TR") + " bölge"; drawMap({ type: "FeatureCollection", features }); } function drawMap(data) { const svg = d3.select("#map"); svg.selectAll("*").remove(); const width = 960; const height = 560; // Yalnızca yüklenen bölgeleri kadraja sığdır. const projection = d3.geoNaturalEarth1() .fitExtent( [[8, 8], [width - 8, height - 8]], data ); const path = d3.geoPath(projection); mapLayer = svg.append("g"); mapLayer.selectAll("path.region") .data(features) .join("path") .attr("class", "region") .attr("data-region-id", f => regionId(f)) .attr("fill", f => regionColor(f)) .attr("d", f => path(f) || "") .on("click", async (event, feature) => { await captureRegion(feature); }) .append("title") .text(f => { const p = f.properties || {}; return p.region_name || p.shapeName || p.name || "Bölge"; }); zoomBehavior = d3.zoom() .scaleExtent([1, 12]) .on("zoom", event => { mapLayer.attr("transform", event.transform); }); svg.call(zoomBehavior); mapReady = true; $("mapStatus").textContent = "Harita hazır"; }
